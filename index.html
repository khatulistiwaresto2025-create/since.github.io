function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); sheet.appendRow([ new Date(), data.name, data.orderType, data.location, data.items, data.total, data.notes ]); return ContentService.createTextOutput(JSON.stringify({"result":"success"})) .setMimeType(ContentService.MimeType.JSON); }
